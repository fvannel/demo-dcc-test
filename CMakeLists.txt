cmake_minimum_required(VERSION 3.13)
option(BUILD_RP2040 "Build firmware using an external Raspberry Pi Pico SDK" OFF)
if(BUILD_RP2040)
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
    endif()
    if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
        message(FATAL_ERROR "Set PICO_SDK_PATH to an installed Raspberry Pi Pico SDK")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(dcc_decoder LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dcc_core src/dcc/decoder.cpp)
target_include_directories(dcc_core PUBLIC include)
target_compile_options(dcc_core PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_RP2040)
    pico_sdk_init()
    add_executable(dcc_firmware src/platform/rp2040/main.cpp src/platform/rp2040/capture.cpp)
    target_link_libraries(dcc_firmware PRIVATE dcc_core pico_stdlib hardware_gpio hardware_sync)
    target_include_directories(dcc_firmware PRIVATE src/platform/rp2040)
    pico_enable_stdio_usb(dcc_firmware 0)
    pico_enable_stdio_uart(dcc_firmware 0)
    pico_add_extra_outputs(dcc_firmware)
else()
    enable_testing()
    add_executable(decoder_tests tests/decoder_tests.cpp)
    target_link_libraries(decoder_tests PRIVATE dcc_core)
    add_test(NAME decoder_tests COMMAND decoder_tests)
endif()
